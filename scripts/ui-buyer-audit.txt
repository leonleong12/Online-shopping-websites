async page => {
  const ok=(condition,message)=>{if(!condition)throw new Error(message)};
  const results=[];
  const nav=name=>page.locator('.main-nav').getByRole('button',{name,exact:true}).click();
  const layout=async name=>{for(const width of [1280,390,320]){await page.setViewportSize({width,height:820});await page.waitForTimeout(120);ok(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth+1),`${name}: overflow at ${width}`);}results.push(name);await page.setViewportSize({width:1280,height:820});};
  await page.locator('.support-admin-messages').getByText('验收回复：您好，已收到',{exact:true}).waitFor();
  await layout('live conversation');
  await page.getByRole('button',{name:'工单处理中心',exact:false}).click();
  await page.locator('.support-ticket-toolbar select').selectOption('CLOSED');
  await page.getByRole('row').filter({hasText:'验收工单'}).getByRole('button',{name:'查看 / 处理'}).click();
  await page.getByLabel('处理结果').waitFor();
  await layout('admin ticket details');
  await nav('钱包 / 储值管理');
  await page.getByRole('tab',{name:'储值码',exact:true}).click();
  const code=await page.locator('.wallet-admin tbody tr').first().locator('code').innerText();
  ok(code.startsWith('NV-'),'code missing');
  await nav('退出登录');
  await page.locator('.header-link').click();
  await page.getByRole('textbox',{name:'用户名',exact:true}).fill('walletui_20261018_buyer');
  await page.getByRole('textbox',{name:'密码',exact:true}).fill('WalletQa!2026Only');
  await page.getByRole('button',{name:'登录 →',exact:true}).click();
  await page.getByRole('button',{name:'打开我的钱包'}).click();
  await page.getByLabel('储值卡 / 兑换码',{exact:true}).fill(code);
  await page.getByRole('button',{name:'立即兑换',exact:true}).click();
  await page.getByText('兑换成功，¥10.00 已到账',{exact:true}).waitFor();
  await layout('buyer wallet');
  await page.screenshot({path:'output/playwright/audit-buyer-wallet.png'});
  await page.getByRole('button',{name:'关闭钱包'}).click();
  await page.locator('.header-link').click();
  const cardOrder=page.locator('.order-card').filter({has:page.locator('.delivered-cards')});
  await cardOrder.getByRole('button',{name:'显示卡密',exact:true}).click();
  await cardOrder.locator('pre').getByText('AUDIT-KEY-A',{exact:false}).waitFor();
  await page.context().grantPermissions(['clipboard-read','clipboard-write']);
  await cardOrder.getByRole('button',{name:'一键复制',exact:true}).click();
  ok((await page.evaluate(()=>navigator.clipboard.readText())).includes('AUDIT-KEY-A'),'card copy failed');
  await cardOrder.getByRole('button',{name:'隐藏卡密',exact:true}).click();
  ok(!(await cardOrder.locator('pre').innerText()).includes('AUDIT-KEY-A'),'card hiding failed');
  await layout('buyer orders');
  await page.getByPlaceholder('搜索你正在寻找的好物').fill('walletui_20261018_card');
  await page.locator('.global-search').getByRole('button',{name:'搜索',exact:true}).click();
  await page.locator('.product-card').getByRole('heading',{name:'walletui_20261018_card',exact:true}).waitFor();
  const product=page.locator('.product-card').filter({has:page.getByRole('heading',{name:'walletui_20261018_card',exact:true})});
  await product.getByRole('button',{name:'加入购物袋 +',exact:true}).click();
  await product.getByRole('button',{name:'增加数量',exact:true}).waitFor();
  await product.getByRole('button',{name:'增加数量',exact:true}).click();
  await page.waitForTimeout(250);ok((await product.locator('.product-quantity').innerText()).includes('2'),'cart increment failed');
  await product.getByRole('button',{name:'减少数量',exact:true}).click();
  await page.locator('.bag-link').click();
  await page.getByRole('button',{name:'去结算 →',exact:true}).click();
  await page.getByLabel('收货人姓名',{exact:true}).fill('验收测试');
  await page.getByLabel('手机号',{exact:true}).fill('13800138000');
  await page.getByLabel('收货地址',{exact:true}).fill('本机隔离测试地址');
  await layout('checkout dialog');
  await page.getByRole('button',{name:'提交订单 →',exact:true}).click();
  await page.getByRole('heading',{name:'订单详情',exact:true}).waitFor();
  await page.locator('.payment-deadline strong').waitFor();
  await layout('pending order and countdown');
  return {passed:results,cardCopy:true,redemption:true,cart:true,checkout:true};
}
