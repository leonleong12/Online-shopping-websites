async page => {
  const results=[];
  const sizes=[1280,930,768,390,320];
  const pages=[['客服与工单','support'],['商品管理','products'],['订单管理','orders'],['钱包 / 储值管理','wallet'],['经营数据','stats'],['首页设置','site'],['所有商品','store']];
  for(const [name,key] of pages){
    await page.setViewportSize({width:1280,height:820});
    await page.locator('.main-nav').getByRole('button',{name,exact:true}).click();
    await page.waitForTimeout(350);
    if(key==='support') await page.locator('.support-settings summary').click();
    for(const theme of ['light','dark']){
      const current=await page.locator('html').getAttribute('data-theme');
      if(current!==theme) await page.locator('.theme-toggle').click();
      for(const width of sizes){
        await page.setViewportSize({width,height:820});
        await page.waitForTimeout(380);
        const size=await page.evaluate(()=>({width:innerWidth,scroll:document.documentElement.scrollWidth,body:document.body.scrollWidth}));
        results.push({page:key,theme,...size});
        if(width===1280||width===390) await page.screenshot({path:`output/playwright/audit-${key}-${theme}-${width}.png`,fullPage:width===390});
      }
    }
  }
  await page.setViewportSize({width:1280,height:820});
  return {checks:results.length,overflow:results.filter(r=>r.scroll>r.width+1),results};
}
